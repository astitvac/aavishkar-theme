Key Remix Features for Aavishkar.ai

1. File-based Routing & Nested Layouts

Remix uses a file-based routing system that maps directly to URL paths.

app/
├── root.tsx             # Root layout (header/footer)
├── routes/
│   ├── _index.tsx       # Home page 
│   ├── notebooks/
│   │   ├── _index.tsx   # /notebooks
│   │   ├── $id.tsx      # /notebooks/:id
│   │   └── create.tsx   # /notebooks/create
│   ├── repository/
│   │   ├── _index.tsx   # /repository
│   │   └── $id.tsx      # /repository/:id
│   └── about.tsx        # /about

2. Data Loading with Loaders

Each route can define a loader function that fetches data before rendering.

// app/routes/notebooks/$id.tsx
export async function loader({ params }) {
  // This runs on the server before rendering
  const notebook = await getNotebook(params.id);
  if (!notebook) {
    throw new Response("Notebook not found", { status: 404 });
  }
  return notebook;
}

export default function NotebookPage() {
  // Data is already available when component renders
  const notebook = useLoaderData();
  
  return (
    <div>
      <h1>{notebook.title}</h1>
      <div>{notebook.content}</div>
    </div>
  );
}

3. Form Handling with Actions

Remix provides built-in form handling that works even without JavaScript.

// app/routes/contact.tsx
export async function action({ request }) {
  const formData = await request.formData();
  const name = formData.get("name");
  const email = formData.get("email");
  const message = formData.get("message");
  
  // Validate
  const errors = {};
  if (!name) errors.name = "Name is required";
  if (!email) errors.email = "Email is required";
  if (!message) errors.message = "Message is required";
  
  if (Object.keys(errors).length) {
    return { errors };
  }
  
  // Process form (e.g., send email)
  await sendContactEmail({ name, email, message });
  return { success: true };
}

export default function Contact() {
  const actionData = useActionData();
  const navigation = useNavigation();
  const busy = navigation.state === "submitting";
  
  return (
    <Form method="post">
      <input name="name" />
      {actionData?.errors?.name && (
        <p className="error">{actionData.errors.name}</p>
      )}
      <input name="email" type="email" />
      {actionData?.errors?.email && (
        <p className="error">{actionData.errors.email}</p>
      )}
      <textarea name="message"></textarea>
      {actionData?.errors?.message && (
        <p className="error">{actionData.errors.message}</p>
      )}
      <button type="submit" disabled={busy}>
        {busy ? "Sending..." : "Send Message"}
      </button>
      {actionData?.success && <p>Message sent successfully!</p>}
    </Form>
  );
}

4. Error Boundaries

Remix provides built-in error handling that prevents your app from crashing.

// Error boundary for a specific route
export function ErrorBoundary() {
  const error = useRouteError();
  
  if (isRouteErrorResponse(error) && error.status === 404) {
    return <div>Notebook not found</div>;
  }
  
  return (
    <div>
      <h1>Error!</h1>
      <p>Something went wrong while loading this notebook.</p>
      <p>{error.message || "Unknown error"}</p>
    </div>
  );
}
